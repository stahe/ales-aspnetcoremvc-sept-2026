    8. Autenticación y autorización

    Para cada solicitud surgen dos preguntas distintas:

    • ¿Quién es el usuario? Esto es la autenticación: el usuario se ha identificado con un nombre de usuario y una contraseña, y cada solicitud posterior debe demostrar que proviene de él;
    • ¿Tiene el usuario permiso para acceder a esta página? Esto es la autorización: en el caso de estudio, solo un usuario con el rol ADMIN gestiona los médicos y los clientes.

    Como el servidor no tiene estado (capítulo 6), la prueba de identidad debe incluirse en cada solicitud. Se utiliza un token JWT ([JSON Web Token]): tras el inicio de sesión, el servidor genera un token que contiene la identidad del usuario y lo firma con una clave secreta. El token se almacena en una cookie; el navegador lo reenvía con cada solicitud; el servidor verifica la firma. El usuario puede leer su token, pero no puede modificarlo: cualquier modificación invalida la firma.

    [ASP.NET Core] distingue claramente las dos cuestiones:

    • la autenticación la gestionan un middleware ([UseAuthentication]) y un esquema de autenticación: en este caso, el esquema [JwtBearer], proporcionado por el paquete [NuGet] [Microsoft.AspNetCore.Authentication.JwtBearer]. En cada solicitud, busca un token, lo verifica y deduce de él el usuario, [HttpContext.User];
    • la autorización la gestiona el atributo [[Authorize]], verificado por el middleware [UseAuthorization]: compara los requisitos de la acción con lo que se sabe del usuario.

    Los ejemplos de este capítulo usan dos paquetes [NuGet]: [Microsoft.AspNetCore.Authentication.JwtBearer] y [BCrypt.Net-Next] (hash de contraseñas). Están declarados en el archivo de proyecto; el comando [dotnet build] (o [dotnet run]) los descarga desde nuget.org durante la primera compilación: por lo tanto, se requiere una conexión a Internet.

    namespace Example25.Auth;

public record User(string Login, string Name, string Role, string PasswordHash);

// los usuarios, en memoria (el próximo capítulo los pondrá en una base de datos)
public class UsersService
{
    private readonly List<User> users =
    [
        // una contraseña NUNCA se guarda, solo su hash
        Create("marie", "Marie Pélissier", "USER", "marie"),
        Create("admin", "Administrator", "ADMIN", "admin"),
    ];

    public User? Find(string login) =>
        users.Find(u => u.Login == login);

    // bcrypt, con un costo de 10: 2^10 rondas de cálculo, deliberadamente lento
    private static User Create(string login, string name, string role, string password) =>
        new(login, name, role, BCrypt.Net.BCrypt.HashPassword(password, 10));
}
    • línea 3: un usuario, junto con su rol;
    • líneas 10–12: una contraseña nunca se almacena; solo se conserva su hash, una huella calculada mediante una función unidireccional. Es posible verificar que una contraseña corresponde a una huella, pero no recuperar la contraseña a partir de la huella;
    • líneas 19–20: [BCrypt.Net.BCrypt.HashPassword(password, 10)] calcula el hash con el algoritmo [bcrypt]. El 10 representa el “costo”: [bcrypt] es deliberadamente lento (2¹⁰ ciclos de cálculo) para disuadir a quien intente millones de contraseñas. Dos hashes de la misma contraseña son diferentes (se agrega una “sal” aleatoria): [BCrypt.Verify] se encarga de verificarlos.

    using System.Text;
using Microsoft.IdentityModel.JsonWebTokens;
using Microsoft.IdentityModel.Tokens;

namespace Example25.Auth;

public class AuthService(UsersService users, IConfiguration configuration)
{
    // el nombre de la cookie que lleva el token
    public const string Cookie = "token";

    // la clave que firma los tokens, leída de appsettings.json (sección Jwt)
    public static SymmetricSecurityKey Key(IConfiguration configuration) =>
        new(Encoding.UTF8.GetBytes(configuration["Jwt:Secret"]!));

    // null si el inicio de sesión es desconocido o la contraseña es incorrecta
    public User? Validate(string? login, string? password)
    {
        var user = users.Find(login ?? "");
        if (user is null || password is null)
        {
            return null;
        }
        return BCrypt.Net.BCrypt.Verify(password, user.PasswordHash) ? user : null;
    }

    // el token JWT del usuario, firmado con la clave secreta
    public string CreateToken(User user)
    {
        var descriptor = new SecurityTokenDescriptor
        {
            // el contenido del token: los "claims" (afirmaciones sobre el usuario)
            Claims = new Dictionary<string, object>
            {
                ["sub"] = user.Login,
                ["name"] = user.Name,
                ["role"] = user.Role,
            },
            Expires = DateTime.UtcNow.AddMinutes(configuration.GetValue<int>("Jwt:DurationMinutes")),
            SigningCredentials = new SigningCredentials(Key(configuration), SecurityAlgorithms.HmacSha256),
        };
        return new JsonWebTokenHandler().CreateToken(descriptor);
    }
}

    using System.Text.Encodings.Web;
using System.Text.Unicode;
using Example25.Auth;
using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.Extensions.FileProviders;
using Microsoft.IdentityModel.Tokens;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddControllersWithViews();
builder.Services.AddSingleton(HtmlEncoder.Create(UnicodeRanges.All));
builder.Services.AddSingleton<UsersService>();
builder.Services.AddSingleton<AuthService>();

// autenticación: un token JWT, verificado en cada solicitud
builder.Services
    .AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(options =>
    {
        options.MapInboundClaims = false; // conservar los nombres de los claims tal cual ("sub", "role"...)
        options.TokenValidationParameters = new TokenValidationParameters
        {
            ValidateIssuerSigningKey = true,
            IssuerSigningKey = AuthService.Key(builder.Configuration), // la firma debe ser la nuestra
            ValidateLifetime = true,                                   // el token no debe estar vencido
            ValidateIssuer = false,
            ValidateAudience = false,
            NameClaimType = "name", // User.Identity.Name
            RoleClaimType = "role", // [Authorize(Roles = ...)], User.IsInRole(...)
        };
        options.Events = new JwtBearerEvents
        {
            // 1. el token se lee de la cookie (y no del encabezado Authorization)
            OnMessageReceived = context =>
            {
                context.Token = context.Request.Cookies[AuthService.Cookie];
                return Task.CompletedTask;
            },
            // 2. página protegida sin un token válido: a la página de inicio de sesión
            OnChallenge = context =>
            {
                context.HandleResponse(); // construimos la respuesta nosotros mismos (sin 401)
                context.Response.Redirect("/login");
                return Task.CompletedTask;
            },
        };
    });
builder.Services.AddAuthorization();

var app = builder.Build();

app.UseStaticFiles(new StaticFileOptions
{
    FileProvider = new PhysicalFileProvider(Path.GetFullPath(Path.Combine(builder.Environment.ContentRootPath, "..", "common", "bootstrap"))),
    RequestPath = "/bootstrap",
});
app.UseAuthentication(); // ¿quién es? -> HttpContext.User
app.UseAuthorization();  // ¿está permitido? -> [Authorize]

app.MapControllers();
app.Run();

    Comentemos este código:

    • líneas 16–17: declaramos el esquema de autenticación [JwtBearer] y lo establecemos como esquema predeterminado;
    • línea 19: de forma predeterminada, [JwtBearer] renombra ciertos claims con nombres largos heredados de las tecnologías de Microsoft ([role] pasaría a ser [http://schemas.microsoft.com/ws/2008/06/identity/claims/role]). Conservamos los nombres del token;
    • Líneas 20–26: las comprobaciones realizadas en cada token: la firma (con nuestra clave) y la fecha de expiración. Nuestros tokens no especifican ni el emisor ([issuer]) ni el destinatario ([audience]), por lo que no los verificamos;
    • líneas 27–28: qué claim contiene el nombre del usuario ([User.Identity.Name]) y cuál contiene su rol ([[Authorize(Roles = ...)]], [User.IsInRole(...)]);
    • líneas 33–36: los eventos de [JwtBearer] permiten personalizar su comportamiento. De forma predeterminada, busca el token en el encabezado [Authorization: Bearer ...] de la solicitud: así funciona un cliente [JavaScript] o una aplicación móvil. Un navegador no agrega este encabezado por sí solo: el token se recupera de la cookie;
    • líneas 39–42: el desafío (challenge) se produce cuando se solicita una página protegida sin un token válido. De forma predeterminada, la respuesta es un error 401 sin contenido: útil para un cliente [JavaScript], pero desconcertante para un usuario. Se redirige al usuario a la página de inicio de sesión. [HandleResponse()] le indica a [JwtBearer] que la respuesta ya está siendo manejada;
    • líneas 56–57: los dos componentes de middleware, en este orden: primero se determina quién es el usuario y luego se verifican sus permisos.

    using Example25.Auth;
using Microsoft.AspNetCore.Mvc;

namespace Example25.Controllers;

public class AuthController(AuthService auth, IConfiguration configuration) : Controller
{
    [HttpGet("/")]
    public IActionResult Home() => Redirect("/profile");

    [HttpGet("login")]
    public IActionResult Form() => View("Login");

    [HttpPost("login")]
    public IActionResult Login(string? login, string? password)
    {
        var user = auth.Validate(login, password);
        if (user is null)
        {
            // un mensaje deliberadamente vago: no decimos si es el inicio de sesión o la contraseña
            ViewData["Error"] = "Incorrect login or password.";
            ViewData["Login"] = login;
            return View("Login");
        }
        // el token, en una cookie que el JavaScript de la página no puede leer
        Response.Cookies.Append(AuthService.Cookie, auth.CreateToken(user), new CookieOptions
        {
            HttpOnly = true,
            SameSite = SameSiteMode.Lax,
            MaxAge = TimeSpan.FromMinutes(configuration.GetValue<int>("Jwt:DurationMinutes")),
        });
        return Redirect("/profile");
    }

    // cerrar sesión cambia el estado del usuario: es un POST
    [HttpPost("logout")]
    public IActionResult Logout()
    {
        Response.Cookies.Delete(AuthService.Cookie);
        return Redirect("/login");
    }
}

    Comentemos este código:

    • líneas 14–23: el inicio de sesión. Si falla, se vuelve a mostrar el formulario con un mensaje deliberadamente vago: decir “este login no existe” ayudaría a un atacante a encontrar logins válidos;
    • líneas 26–32: si tiene éxito, el token se almacena en una cookie [HttpOnly]; el [JavaScript] de la página no puede leerla, lo que protege el token si se ejecutara un script malicioso en la página. [SameSite = Lax]: la cookie no se incluye en las solicitudes POST enviadas desde otro sitio, lo que protege contra la falsificación de solicitudes ([CSRF]). La duración de la cookie coincide con la del token;
    • líneas 36–40: cerrar sesión elimina la cookie ([Cookies.Delete]). Se trata de un POST: modifica el estado del usuario (vea el ejemplo 21).

    using System.Collections.Concurrent;

namespace Example27.Auth;

// un captcha: su identificador y su imagen (SVG)
public record Captcha(string Id, string Svg);

// singleton: los captchas pendientes se guardan en la memoria del servidor
public class CaptchaService
{
    private static readonly TimeSpan Lifetime = TimeSpan.FromMinutes(10);

    // identificador -> (respuesta esperada, fecha de vencimiento).
    // ConcurrentDictionary: un diccionario utilizable por varias solicitudes simultáneas
    private readonly ConcurrentDictionary<string, (string Text, DateTime Expires)> pending = new();

    public Captcha Generate()
    {
        var text = CaptchaDrawing.RandomText(5);
        var id = Guid.NewGuid().ToString(); // un identificador aleatorio, imposible de adivinar
        pending[id] = (text, DateTime.UtcNow + Lifetime);
        return new Captcha(id, CaptchaDrawing.Svg(text));
    }

    // el captcha se CONSUME en todos los casos: una imagen se usa una sola vez
    public bool Check(string? id, string? input)
    {
        if (id is null || input is null || !pending.TryRemove(id, out var expected))
        {
            return false;
        }
        return expected.Expires > DateTime.UtcNow
            && string.Equals(expected.Text, input.Trim(), StringComparison.OrdinalIgnoreCase);
    }
}

    Analicemos este código:

    • línea 15: el servidor almacena las respuestas esperadas bajo un identificador aleatorio. La página solo contiene la imagen y el identificador: la respuesta nunca sale del servidor. El servicio es un singleton, utilizado por solicitudes concurrentes: [ConcurrentDictionary] es un diccionario diseñado para este fin;
    • líneas 17–22: se elige al azar una cadena de cinco caracteres, se almacena y se dibuja como una imagen. El dibujo está a cargo de la clase [CaptchaDrawing] (archivo [Auth/CaptchaDrawing.cs], tomada del caso de estudio y comentada línea por línea): cada carácter es un conjunto de líneas que se agranda, se inclina y se distorsiona al azar, y la imagen se difumina con tres curvas. Se excluyen los caracteres que se confunden fácilmente (0 y O, 1, I y L);
    • líneas 26–33: la verificación consume el CAPTCHA ([TryRemove] lo elimina del diccionario): una imagen se usa una sola vez; un bot no puede reutilizar una solución que ya encontró. Las mayúsculas no cuentan.

    namespace Example27.Auth;

// singleton: cuenta los fallos de inicio de sesión de cada dirección IP
public class LimiterService
{
    private const int MaxFailures = 3;
    private static readonly TimeSpan BlockDuration = TimeSpan.FromMinutes(1);

    private class Counter
    {
        public int Failures;
        public DateTime BlockedUntil;
    }

    private readonly Dictionary<string, Counter> counters = [];
    private readonly Lock padlock = new(); // varias solicitudes simultáneas: un acceso a la vez

    // el número de segundos de bloqueo restantes (0: no bloqueado)
    public int BlockedSeconds(string ip)
    {
        lock (padlock)
        {
            if (!counters.TryGetValue(ip, out var c)) return 0;
            var remaining = c.BlockedUntil - DateTime.UtcNow;
            return remaining > TimeSpan.Zero ? (int)Math.Ceiling(remaining.TotalSeconds) : 0;
        }
    }

    public void Failure(string ip)
    {
        lock (padlock)
        {
            if (!counters.TryGetValue(ip, out var c))
            {
                c = new Counter();
                counters[ip] = c;
            }
            c.Failures++;
            if (c.Failures >= MaxFailures)
            {
                c.BlockedUntil = DateTime.UtcNow + BlockDuration;
                c.Failures = 0; // después del bloqueo, se empieza de nuevo desde cero
            }
        }
    }

    public void Success(string ip)
    {
        lock (padlock)
        {
            counters.Remove(ip);
        }
    }
}
    • líneas 6–7: tres intentos fallidos bloquean la dirección IP durante un minuto;
    • línea 16: el diccionario de contadores es compartido por todas las solicitudes. La instrucción [lock (padlock) { ... }] garantiza que una sola solicitud a la vez ejecute el bloque: dos fallos simultáneos no pueden perderse (vea [Interlocked], capítulo 6). [Lock] es el tipo de bloqueo a partir de [.NET] 9 y versiones posteriores;
    • líneas 19–25: el tiempo de bloqueo restante;
    • líneas 29–42: se cuenta un fallo; al tercer fallo, se bloquea la dirección;
    • líneas 47–51: un éxito restablece el contador a cero.

    using Example27.Auth;
using Microsoft.AspNetCore.Mvc;

namespace Example27.Controllers;

public class AuthController(AuthService auth, CaptchaService captchas, LimiterService limiter,
    IConfiguration configuration) : Controller
{
    [HttpGet("/")]
    public IActionResult Home() => Redirect("/profile");

    [HttpGet("login")]
    public IActionResult Form() => LoginPage(null, null);

    [HttpPost("login")]
    public IActionResult Login(string? login, string? password, string? captchaId, string? captcha)
    {
        // la dirección IP del cliente
        var ip = HttpContext.Connection.RemoteIpAddress?.ToString() ?? "?";

        // 1. una dirección bloqueada ni siquiera se atiende
        var seconds = limiter.BlockedSeconds(ip);
        if (seconds > 0)
        {
            return LoginPage(login, $"Too many attempts: try again in {seconds} s.");
        }
        // 2. primero el captcha (barato), luego la contraseña (bcrypt, costoso)
        if (!captchas.Check(captchaId, captcha))
        {
            limiter.Failure(ip);
            return LoginPage(login, "The image code is incorrect.");
        }
        var user = auth.Validate(login, password);
        if (user is null)
        {
            limiter.Failure(ip);
            return LoginPage(login, "Incorrect login or password.");
        }
        limiter.Success(ip);
        Response.Cookies.Append(AuthService.Cookie, auth.CreateToken(user), new CookieOptions
        {
            HttpOnly = true,
            SameSite = SameSiteMode.Lax,
            MaxAge = TimeSpan.FromMinutes(configuration.GetValue<int>("Jwt:DurationMinutes")),
        });
        return Redirect("/profile");
    }

    [HttpPost("logout")]
    public IActionResult Logout()
    {
        Response.Cookies.Delete(AuthService.Cookie);
        return Redirect("/login");
    }

    // cada visualización del formulario construye un captcha NUEVO
    private ViewResult LoginPage(string? login, string? error)
    {
        ViewData["Login"] = login;
        ViewData["Error"] = error;
        return View("Login", captchas.Generate());
    }
}

    Comentemos este código:

    • línea 19: la dirección IP del cliente, proporcionada por la conexión de red;
    • líneas 22–25: una dirección bloqueada ni siquiera es atendida;
    • líneas 28–31: primero se verifica el CAPTCHA y luego la contraseña (una operación [bcrypt] costosa en términos de cómputo) solo si el CAPTCHA es correcto. Un CAPTCHA incorrecto cuenta como un fallo;
    • líneas 57–61: cada vez que se muestra el formulario, se genera un nuevo CAPTCHA, que se convierte en el modelo de la vista.
